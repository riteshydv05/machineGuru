# ============================================================
# MachineGuru Backend — Python Dependencies
# ============================================================
# Target: Ubuntu 20.04/22.04, Python 3.10+, ARM64 (Jetson Orin)
#
# Design principle: MINIMAL install for space-constrained devices.
#
# Heavy AI packages (torch, sentence-transformers, numpy) are
# intentionally EXCLUDED because on Jetson Orin:
#   - torch    → installed by JetPack/jetson_setup.sh (CUDA wheel)
#   - sentence-transformers → not needed when USE_OLLAMA_EMBEDDINGS=true
#   - numpy    → installed by JetPack as a system package
#
# Set USE_OLLAMA_EMBEDDINGS=true in .env to use Ollama's
# /api/embeddings endpoint instead of SentenceTransformers.
# This avoids installing torch, sentence-transformers, and numpy
# from PyPI entirely — saving ~2 GB of disk space on the Jetson.
#
# On x86_64 dev machines that DO need SentenceTransformers locally,
# install the extras manually:
#   pip install sentence-transformers torch
# ============================================================

# ── Core Web Framework ──────────────────────────────────────
fastapi==0.115.6
uvicorn[standard]==0.34.0
pydantic==2.10.4
pydantic-settings==2.7.1
sse-starlette==2.2.1

# ── Vector Database ─────────────────────────────────────────
# qdrant-client: use [fastembed] extras removed to avoid heavy deps on Jetson
qdrant-client>=1.7.0,<2.0.0
httpx>=0.28.0,<1.0.0

# ── Document Processing ──────────────────────────────────────
# pymupdf: use a version with pre-built ARM64 wheels to avoid source compilation on Jetson.
# If this fails: pip install --no-cache-dir pymupdf --only-binary=:all:
PyMuPDF>=1.23.0,<1.26.0
python-docx==1.1.2

# ── HTTP client (sync, for Ollama REST calls in scripts) ────
requests>=2.31.0,<3.0.0

# ── Hybrid Retrieval ─────────────────────────────────────────
rank-bm25>=0.2.2,<1.0.0

# ── Image processing (needed by image_extractor.py) ─────────
# Pillow is lightweight (~4 MB) — kept.
# pytesseract is OPTIONAL: image_extractor.py catches ImportError
# and silently skips OCR if it is not installed.
Pillow>=10.0.0,<12.0.0

# ── Monitoring ──────────────────────────────────────────────
# psutil is tiny and used for memory/CPU tracking.
# prometheus-client is OPTIONAL: main.py catches ImportError
# and disables the /metrics endpoint if not installed.
psutil>=6.1.0,<7.0.0

# ── Utilities ────────────────────────────────────────────────
python-multipart==0.0.20
python-dotenv==1.0.1
loguru==0.7.3

# ── NOT installed here — see notes above ────────────────────
# torch               → JetPack wheel (jetson_setup.sh) or manual install
# sentence-transformers → not needed with USE_OLLAMA_EMBEDDINGS=true
# numpy               → JetPack system package or pulled in by torch
# pytesseract         → optional OCR, install manually if needed
# prometheus-client   → optional metrics, install manually if needed
